## 二、数据模型

### 2.1 目标模型 (Goal)

```typescript
interface Goal {
  id: string;                         // 唯一标识
  title: string;                      // 目标名称
  description: string;                // 目标描述
  level: 1 | 2 | 3 | 4;              // 层级（1=人生, 2=阶段, 3=年度, 4=短期）
  parentId: string | null;            // 父目标 ID
  status: 'active' | 'completed' | 'abandoned';
  progress: number;                   // 进度百分比 (0-100)
  progressAuto: boolean;              // 是否自动计算进度
  priority: 1 | 2 | 3;               // 优先级
  weight: number;                     // 权重（用于汇总计算）
  startDate: string;                  // 开始日期 (YYYY-MM-DD)
  dueDate: string | null;             // 截止日期
  tags: string[];                     // 标签（用于关联）
  filePath: string;                   // 目标文件路径
  createdAt: string;                  // 创建时间
  updatedAt: string;                  // 更新时间
}

interface GoalTree {
  goal: Goal;
  children: GoalTree[];
  aggregatedProgress: number;         // 加权汇总进度
}
```

### 2.2 任务模型 (Task)

```typescript
interface Task {
  id: string;                         // 唯一标识
  title: string;                      // 任务标题
  status: 'pending' | 'in-progress' | 'completed' | 'cancelled';
  priority: 1 | 2 | 3 | 4 | 5;       // 优先级 (1=最高)
  dueDate: string | null;             // 截止日期
  relatedGoalId: string | null;       // 关联目标 ID
  tags: string[];                     // 标签（#目标/xxx #任务/xxx）
  isNoteworthy: boolean;              // 是否标记为重要
  filePath: string;                   // 所在文件路径
  lineNumber: number;                 // 文件行号
  createdAt: string;                  // 创建时间
  completedAt: string | null;         // 完成时间
}
```

### 2.3 笔记模型 (Note)

```typescript
interface Note {
  id: string;                         // 唯一标识
  type: 'daily' | 'weekly' | 'monthly' | 'yearly' | 'goal';
  title: string;                      // 标题
  date: string;                       // 日期/周期标识
  filePath: string;                   // 文件路径
  createdAt: string;                  // 创建时间
  updatedAt: string;                  // 更新时间
}

interface NoteworthyItem {
  id: string;                         // 唯一标识
  content: string;                    // 原文内容
  sourceType: 'daily' | 'weekly' | 'monthly' | 'yearly' | 'goal';
  sourcePath: string;                 // 来源文件路径
  sourceDate: string;                 // 来源日期
  tags: string[];                     // 标签
  createdAt: string;                  // 创建时间
}
```

### 2.4 标签模型 (Tag)

```typescript
interface TagDefinition {
  tag: string;                        // 标签名（如：目标/健康）
  type: 'goal' | 'task' | 'category' | 'noteworthy';
  description: string;                // 标签描述
  color: string;                      // 标签颜色
  createdAt: string;                  // 创建时间
}

interface TagUsage {
  tag: string;                        // 标签名
  count: number;                      // 使用次数
  lastUsed: string;                   // 最后使用时间
  sources: string[];                  // 来源文件列表
}
```

## 六、API 设计

### 6.1 核心 API 接口

```typescript
// GoalManager API
interface IGoalManager {
  createGoal(data: CreateGoalDTO): Promise<Goal>;
  updateGoal(id: string, data: UpdateGoalDTO): Promise<Goal>;
  deleteGoal(id: string): Promise<void>;
  getGoal(id: string): Promise<Goal | null>;
  getGoalsByLevel(level: number): Promise<Goal[]>;
  getGoalTree(): Promise<GoalTree>;
  updateProgress(id: string, progress: number): Promise<Goal>;
  calculateProgress(id: string): Promise<number>;
}

// NoteManager API
interface INoteManager {
  createNote(type: NoteType, date: string): Promise<Note>;
  loadNote(path: string): Promise<Note>;
  updateNote(path: string, content: string): Promise<void>;
  deleteNote(path: string): Promise<void>;
  getDailyNote(date: string): Promise<Note | null>;
  getWeeklyNote(week: string): Promise<Note | null>;
  getMonthlyNote(month: string): Promise<Note | null>;
  getYearlyNote(year: string): Promise<Note | null>;
}

// NoteAggregator API
interface INoteAggregator {
  aggregateDailyToWeekly(weekNotePath: string): Promise<NoteworthyItem[]>;
  aggregateWeeklyToMonthly(monthNotePath: string): Promise<NoteworthyItem[]>;
  aggregateMonthlyToYearly(yearNotePath: string): Promise<NoteworthyItem[]>;
  aggregateToGoalNote(goalNotePath: string, filters?: TagFilter[]): Promise<NoteworthyItem[]>;
  generateAggregatedContent(items: NoteworthyItem[]): string;
}

// TaskManager API
interface ITaskManager {
  scanTasks(content: string, filePath: string): Task[];
  getTasksByTag(tag: string): Task[];
  getTasksByGoal(goalId: string): Task[];
  getTodayTasks(): Task[];
  getOverdueTasks(): Task[];
}

// TagParser API
interface ITagParser {
  parseTags(content: string): ParsedLine[];
  extractTags(line: string): string[];
  extractGoalTags(content: string): string[];
  extractNoteworthyLines(content: string): ParsedLine[];
}

// DashboardManager API
interface IDashboardManager {
  getOverview(): Promise<DashboardOverview>;
  getGoalProgress(): Promise<GoalProgress[]>;
  getWeeklyHeatmap(year: number): Promise<HeatmapData>;
  getNoteworthyRecent(days: number): Promise<NoteworthyItem[]>;
  refresh(): void;
}
```

### 6.2 事件系统

```typescript
// 事件类型
type LifeManagerEvent = 
  | 'goal:created'
  | 'goal:updated'
  | 'goal:deleted'
  | 'goal:progress-changed'
  | 'task:completed'
  | 'task:created'
  | 'note:created'
  | 'note:updated'
  | 'noteworthy:aggregated'
  | 'dashboard:refresh';

// 事件监听示例
plugin.on('task:completed', (task: Task) => {
  if (task.relatedGoalId) {
    const newProgress = goalManager.calculateProgress(task.relatedGoalId);
    goalManager.updateProgress(task.relatedGoalId, newProgress);
  }
  dashboardManager.refresh();
});
```

// Goals.test.ts
describe('GoalManager', () => {

      taskTracker.scanTasks(`

        - [ ] 任务2 #目标/测试目标
      `, 'test.md');
      
      const progress = await goalManager.calculateProgress(goal.id);
      expect(progress).toBe(50); // 1/2 完成
    });
  });
});

// Tags.test.ts
describe('TagParser', () => {
